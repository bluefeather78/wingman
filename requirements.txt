# Web service (app/) — shipped to Render.
#
# The offline agent toolkit at the repo root is still stdlib-only (http.server was
# dropped from the *web* layer only); those scripts add no third-party deps here.
# Phase 1 of the rearchitecture (PLAN_1_decompose.md) moved the public web layer from
# a hand-rolled http.server monolith to FastAPI + uvicorn.
fastapi>=0.115
uvicorn[standard]>=0.30

# Session auth (Phase 2, PLAN_2_auth.md): PyJWT signs/verifies the access+refresh tokens;
# argon2-cffi hashes the (already client-SHA-256'd) password server-side. Both ship Linux
# wheels, so Render installs them without a compiler.
PyJWT>=2.8
argon2-cffi>=23.1

# numpy: the recall stage of the curated-matching pipeline (app/services/matching.py,
# OPPORTUNITY_MATCHING_PLAN.md Phase 5) scores ~1,500 catalog embeddings against a student's
# per-theme embeddings by cosine similarity on every search. A naive Python triple loop over
# rows x themes x ~768 dims runs ~1s; numpy/BLAS makes it ~milliseconds. numpy ships manylinux
# wheels, so Render installs it without a compiler (same as PyJWT/argon2 above).
numpy>=1.26

# PyPDF2: the resume/LinkedIn import endpoint AND the action-item substrate's capture layer
# (source_capture.py — decodes the base64 PDF bytes Claude web_fetch returns, so a program's
# guidelines PDF can be read and verified). Both degrade gracefully if it is missing (resume
# falls back to plain text; capture yields no text for a PDF, so that source backs nothing),
# so it is not a hard dep — but the Render service should install it so PDFs parse.
# python-docx is resume-only.
PyPDF2>=3.0
python-docx>=1.1
